#include <stdio.h>
#include <stdlib.h>
void VerMatriz(int **, int f, int c);
int main()
{  int **Dinamica, N, M;
   int i, j;
   printf("Ingrese N y M: ");
   scanf("%d %d", &N, &M);

   /// reserva de memoria
   Dinamica = malloc(N * sizeof(int *));
   for (i=0;i < N; i++)
      Dinamica[i] = malloc(M * sizeof(int));

   /// cargar la matriz
   for (i=0;i < N; i++)
     for (j=0; j< M; j++)
          Dinamica[i][j] = i*M + j + 1;

   VerMatriz(Dinamica, N,M);

   /// Liberando la memoria
   for (i=0;i < N; i++)
       free(Dinamica[i]);
   free(Dinamica);
   return(0);
}
void VerMatriz(int **Mat, int filas, int cols)
{ int i,j;

  for (i=0;i < filas; i++){
     for (j=0; j<cols; j++)
       printf("%2d  ",Mat[i][j]);
     printf("\n");
  }

}
